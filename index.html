<head>

  <meta charset="UTF-8">

  <meta name="viewport" content="width=device-width, initial-scale=1.0">

  <title>中文寫作小冒險</title>

  <script src="https://cdn.tailwindcss.com"></script>

  <script src="https://unpkg.com/react@18/umd/react.production.min.js"></script>

  <script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>

  <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>

  <link href="https://fonts.googleapis.com/css2?family=Noto+Sans+TC:wght@400;500;700&display=swap" rel="stylesheet">

  <style>

    body { font-family: 'Noto Sans TC', sans-serif; background-color: #F3F4F6; }

  </style>

</head>

<body>

  <div id="root"></div>


  <script type="text/babel">

    const { useState, useEffect } = React;


    function App() {

      const [apiKey, setApiKey] = useState(localStorage.getItem('gemini_api_key') || '');

      const [showSettings, setShowSettings] = useState(!localStorage.getItem('gemini_api_key'));

      

      const [gameState, setGameState] = useState('setup');

      const [turn, setTurn] = useState(1);

      const [loading, setLoading] = useState(false);


      const [topicType, setTopicType] = useState('旅行');

      const [location, setLocation] = useState('');

      const [characters, setCharacters] = useState('');

      const [goal, setGoal] = useState('');


      const [history, setHistory] = useState([]);

      const [feedback, setFeedback] = useState('歡迎來到寫作冒險！準備好和我一起創作專屬的故事了嗎？');

      const [currentStory, setCurrentStory] = useState('');

      const [challenge, setChallenge] = useState(null);

      const [userInput, setUserInput] = useState('');

      const [usedWords, setUsedWords] = useState([]);


      const saveApiKey = (key) => {

        setApiKey(key);

        localStorage.setItem('gemini_api_key', key);

        setShowSettings(false);

      };


      // 呼叫 Gemini API 的核心函數

      const callGeminiAI = async (userPrompt, systemPrompt) => {

        const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, {

          method: 'POST',

          headers: { 'Content-Type': 'application/json' },

          body: JSON.stringify({

            contents: [

              { role: 'user', parts: [{ text: systemPrompt + "\n\n" + userPrompt }] }

            ],

            generationConfig: { responseMimeType: "application/json" }

          })

        });


        const data = await res.json();

        if (data.error) {

          throw new Error(data.error.message);

        }


        let rawText = data.candidates[0].content.parts[0].text;

        // 清洗可能存在的 markdown code blocks

        rawText = rawText.replace(/```json/g, '').replace(/```/g, '').trim();

        return JSON.parse(rawText);

      };


      const handleStart = async () => {

        if (!apiKey) {

          alert('請先設定 Gemini API Key！');

          setShowSettings(true);

          return;

        }

        if (!location || !characters || !goal) {

          alert('請填寫完整的故事背景資訊喔！');

          return;

        }


        setLoading(true);

        try {

          const systemPrompt = `

你是一位專為國小生設計的中文寫作遊戲引導員，同時扮演「故事說書人」與「溫柔的語文教練（精靈角色）」。

【核心原則】

1. 全程必須使用規範的「中文書面語」，絕對不能出現任何粵語方言字詞（如：嘅、佢、咁、睇、咗等）。

2. 語氣生動親切，以鼓勵為主。

【故事背景設定】

- 主題類型：${topicType}

- 故事地點：${location}

- 登場人物：${characters}

- 主要目標：${goal}


請嚴格輸出 JSON 格式：

{

  "language_feedback": "歡迎語與鼓勵",

  "story_continuation": "故事的開頭段落（80-150字）",

  "next_challenge": {

    "situation": "情境說明",

    "task": "寫作任務指令",

    "prompt_words": ["詞彙1", "詞彙2", "連接詞"]

  }

}

`;

          const userPrompt = "玩家剛剛做好了故事背景設定。請撰寫故事的開頭段落（story_continuation），並給予第 1 回合的寫作任務（next_challenge）。";

          

          const data = await callGeminiAI(userPrompt, systemPrompt);

          setCurrentStory(data.story_continuation);

          setChallenge(data.next_challenge);

          setFeedback(data.language_feedback);

          setGameState('playing');

        } catch (err) {

          alert('生成失敗：' + err.message);

        }

        setLoading(false);

      };


      const handleSubmitTurn = async () => {

        if (!userInput.trim()) return;

        setLoading(true);


        const newHistory = [...history, { turn, input: userInput, story: currentStory }];

        setHistory(newHistory);


        if (challenge && challenge.prompt_words) {

          challenge.prompt_words.forEach(word => {

            if (userInput.includes(word) && !usedWords.includes(word)) {

              setUsedWords(prev => [...prev, word]);

            }

          });

        }


        try {

          const systemPrompt = `

你是一位專為國小生設計的中文寫作遊戲引導員，同時扮演「故事說書人」與溫柔的語文教練。

【核心原則】全規範中文書面語，絕不使用粵語。

【背景】主題:${topicType} | 地點:${location} | 人物:${characters} | 目標:${goal} | 當前第${turn}回合。


請嚴格輸出 JSON 格式：

{

  "language_feedback": "精靈語文點評（讚賞優點或溫柔指導語病）",

  "story_continuation": "將玩家句子順暢融入後推進的故事段落（80-150字）",

  "next_challenge": {

    "situation": "下一個情境",

    "task": "具體寫作任務",

    "prompt_words": ["詞彙1", "詞彙2", "連接詞"]

  }

}

`;

          const userPrompt = `當前為第 ${turn} 回合。玩家輸入的句子是：「${userInput}」。請分析並推進故事。`;


          const data = await callGeminiAI(userPrompt, systemPrompt);

          setCurrentStory(data.story_continuation);

          setFeedback(data.language_feedback);


          if (turn >= 5) {

            setGameState('ending');

          } else {

            setChallenge(data.next_challenge);

            setTurn(turn + 1);

            setUserInput('');

          }

        } catch (err) {

          alert('提交失敗：' + err.message);

        }

        setLoading(false);

      };


      const handleChipClick = (word) => {

        setUserInput(prev => prev ? `${prev} ${word}` : word);

      };


      return (

        <div className="max-w-3xl mx-auto p-4 min-h-screen flex flex-col justify-between">

          <header className="text-center py-4 border-b border-gray-200 mb-4 relative">

            <h1 className="text-2xl font-bold text-indigo-600">✍️ 中文寫作小冒險</h1>

            <p className="text-sm text-gray-500">專屬你的互動式故事創作集</p>

            <button 

              onClick={() => setShowSettings(!showSettings)}

              className="absolute right-0 top-4 text-xs bg-gray-200 px-3 py-1 rounded-lg text-gray-700 hover:bg-gray-300"

            >

              ⚙️ 設定 API Key

            </button>

          </header>


          {/* API Key 設定彈窗 */}

          {showSettings && (

            <div className="bg-amber-50 border border-amber-200 p-4 rounded-2xl mb-4 space-y-2">

              <h3 className="font-bold text-amber-900 text-sm">設定 Google Gemini API Key</h3>

              <p className="text-xs text-amber-700">為了讓網頁能直接與 AI 溝通，請輸入你的 Gemini API Key（只需設定一次，會安全保存在你的瀏覽器中）：</p>

              <div className="flex gap-2">

                <input

                  type="password"

                  placeholder="貼上你的 API Key (AIzaSy...)"

                  defaultValue={apiKey}

                  id="apiKeyInput"

                  className="flex-1 p-2 border border-amber-300 rounded-xl text-sm bg-white"

                />

                <button

                  onClick={() => saveApiKey(document.getElementById('apiKeyInput').value.trim())}

                  className="px-4 py-2 bg-amber-600 text-white text-sm font-bold rounded-xl hover:bg-amber-700"

                >

                  儲存

                </button>

              </div>

            </div>

          )}


          {gameState === 'setup' && (

            <div className="bg-white p-6 rounded-2xl shadow-sm border border-gray-100 space-y-4">

              <h2 className="text-xl font-bold text-gray-800">第一步：設定你的故事背景</h2>

              

              <div>

                <label className="block font-medium text-gray-700 mb-1">1. 選擇寫作主題：</label>

                <div className="flex gap-2">

                  {['冒險', '旅行', '夢境'].map(t => (

                    <button

                      key={t}

                      onClick={() => setTopicType(t)}

                      className={`px-4 py-2 rounded-xl text-sm font-medium border ${topicType === t ? 'bg-indigo-600 text-white border-indigo-600' : 'bg-gray-50 text-gray-600'}`}

                    >

                      {t === '冒險' ? '🗺️ 一場冒險' : t === '旅行' ? '✈️ 一次旅行' : '🌙 一個夢'}

                    </button>

                  ))}

                </div>

              </div>


              <div>

                <label className="block font-medium text-gray-700 mb-1">2. 故事發生在哪裡？</label>

                <input

                  type="text"

                  placeholder="例如：日本主題樂園、神祕的森林、阿嬤家"

                  value={location}

                  onChange={e => setLocation(e.target.value)}

                  className="w-full p-3 border border-gray-200 rounded-xl focus:outline-none focus:ring-2 focus:ring-indigo-500"

                />

              </div>


              <div>

                <label className="block font-medium text-gray-700 mb-1">3. 有誰一起參與？</label>

                <input

                  type="text"

                  placeholder="例如：我和爸爸、還有一個膽小的朋友"

                  value={characters}

                  onChange={e => setCharacters(e.target.value)}

                  className="w-full p-3 border border-gray-200 rounded-xl focus:outline-none focus:ring-2 focus:ring-indigo-500"

                />

              </div>


              <div>

                <label className="block font-medium text-gray-700 mb-1">4. 最想完成的一件事是什麼？</label>

                <input

                  type="text"

                  placeholder="例如：買到限量版玩具、解開古老謎題"

                  value={goal}

                  onChange={e => setGoal(e.target.value)}

                  className="w-full p-3 border border-gray-200 rounded-xl focus:outline-none focus:ring-2 focus:ring-indigo-500"

                />

              </div>


              <button

                onClick={handleStart}

                disabled={loading}

                className="w-full py-3 bg-indigo-600 text-white font-bold rounded-xl shadow-md hover:bg-indigo-700 transition"

              >

                {loading ? '精靈正在準備故事...' : '開啟寫作之旅 🚀'}

              </button>

            </div>

          )}


          {gameState === 'playing' && (

            <div className="space-y-4 flex-1 flex flex-col justify-between">

              <div className="bg-amber-50 border border-amber-200 p-4 rounded-2xl flex items-start gap-3">

                <span className="text-2xl">🧚‍♂️</span>

                <div>

                  <h3 className="font-bold text-amber-800 text-sm">寫作精靈的回饋：</h3>

                  <p className="text-amber-900 text-sm mt-1">{feedback}</p>

                </div>

              </div>


              <div className="bg-white p-6 rounded-2xl border border-gray-100 shadow-sm space-y-3">

                <div className="flex justify-between text-xs text-gray-400 border-b pb-2">

                  <span>寫作主題：{topicType}</span>

                  <span>回合：{turn} / 5</span>

                </div>

                <p className="text-gray-800 leading-relaxed text-base">{currentStory}</p>

              </div>


              {challenge && (

                <div className="bg-indigo-50 border border-indigo-100 p-5 rounded-2xl space-y-3">

                  <div className="text-sm font-bold text-indigo-900">🎯 當前任務：{challenge.situation}</div>

                  <p className="text-sm text-indigo-700">{challenge.task}</p>


                  <div className="flex flex-wrap gap-2 items-center pt-1">

                    <span className="text-xs text-indigo-500 font-medium">靈感詞庫（點擊使用）：</span>

                    {challenge.prompt_words?.map((word, idx) => (

                      <button

                        key={idx}

                        onClick={() => handleChipClick(word)}

                        className="px-3 py-1 bg-white border border-indigo-200 text-indigo-700 text-xs rounded-full hover:bg-indigo-100 transition shadow-sm"

                      >

                        + {word}

                      </button>

                    ))}

                  </div>


                  <textarea

                    rows="3"

                    placeholder="在這裡輸入你的句子，練習寫規範的中文書面語..."

                    value={userInput}

                    onChange={e => setUserInput(e.target.value)}

                    className="w-full p-3 border border-gray-200 rounded-xl focus:outline-none focus:ring-2 focus:ring-indigo-500 text-sm"

                  ></textarea>


                  <button

                    onClick={handleSubmitTurn}

                    disabled={loading || !userInput.trim()}

                    className="w-full py-3 bg-indigo-600 text-white font-bold rounded-xl hover:bg-indigo-700 disabled:opacity-50 transition"

                  >

                    {loading ? '精靈正在閱讀中...' : turn === 5 ? '完成故事並匯出小書 📖' : '送出句子，繼續冒險 ✨'}

                  </button>

                </div>

              )}

            </div>

          )}


          {gameState === 'ending' && (

            <div className="bg-white p-6 rounded-2xl border border-gray-100 shadow-sm space-y-6 text-center">

              <div className="text-4xl">🎉</div>

              <h2 className="text-2xl font-bold text-gray-800">恭喜完成你的專屬小書！</h2>

              

              <div className="text-left bg-amber-50/50 p-6 rounded-2xl border border-amber-100 space-y-4">

                <h3 className="text-lg font-bold text-amber-900 text-center">《{location}的{topicType}記》</h3>

                {history.map((item, idx) => (

                  <div key={idx} className="space-y-1 text-sm text-gray-700">

                    <p className="italic text-gray-500">＞ {item.input}</p>

                    <p className="leading-relaxed">{item.story}</p>

                  </div>

                ))}

              </div>


              <div className="bg-indigo-50 p-4 rounded-xl text-left space-y-2">

                <h4 className="font-bold text-indigo-900 text-sm">🏆 本次冒險解鎖的好詞庫：</h4>

                <div className="flex flex-wrap gap-2">

                  {usedWords.length > 0 ? (

                    usedWords.map((w, i) => (

                      <span key={i} className="px-3 py-1 bg-white text-indigo-700 text-xs rounded-full border border-indigo-200">{w}</span>

                    ))

                  ) : (

                    <span className="text-xs text-gray-500">繼續加油，下次試著多使用靈感標籤喔！</span>

                  )}

                </div>

              </div>


              <button

                onClick={() => window.location.reload()}

                className="px-6 py-3 bg-indigo-600 text-white font-bold rounded-xl shadow-md hover:bg-indigo-700 transition"

              >

                再玩一次，創作新故事 🔄

              </button>

            </div>

          )}

        </div>

      );

    }


    ReactDOM.createRoot(document.getElementById('root')).render(<App />);

  </script>

</body>

</html>
